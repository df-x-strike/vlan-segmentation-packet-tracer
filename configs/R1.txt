! ==========================================================
! R1 - Passerelle / routage inter-VLAN / ACL / NAT  (Cisco 2911)
! Mots de passe : DEMO DE LABORATOIRE uniquement, a changer.
! ==========================================================
enable
configure terminal
hostname R1
no ip domain-lookup
service password-encryption
enable secret Lab@Secret2026
banner motd # Acces reserve au personnel autorise #
!
! ---------- ACL de filtrage inter-VLAN (une par VLAN, en entree) ----------
ip access-list extended ACL-VLAN10-DIRECTION
 permit udp any eq 68 any eq 67
 permit icmp any any echo-reply
 permit ip 192.168.10.0 0.0.0.255 192.168.30.0 0.0.0.255
 permit ip 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255
 deny ip 192.168.10.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip 192.168.10.0 0.0.0.255 any
!
ip access-list extended ACL-VLAN20-EMPLOYES
 permit udp any eq 68 any eq 67
 permit icmp any any echo-reply
 permit icmp 192.168.20.0 0.0.0.255 host 192.168.30.10 echo
 permit udp 192.168.20.0 0.0.0.255 host 192.168.30.10 eq 53
 permit tcp 192.168.20.0 0.0.0.255 host 192.168.30.10 eq 80
 permit tcp 192.168.20.0 0.0.0.255 host 192.168.30.10 eq 443
 permit tcp 192.168.20.0 0.0.0.255 host 192.168.30.10 eq 21
 deny ip 192.168.20.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip 192.168.20.0 0.0.0.255 any
!
ip access-list extended ACL-VLAN30-SERVEURS
 permit icmp any any echo-reply
 permit tcp 192.168.30.0 0.0.0.255 any established
 permit udp 192.168.30.0 0.0.0.255 eq 53 any
 deny ip 192.168.30.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip 192.168.30.0 0.0.0.255 any
!
ip access-list extended ACL-VLAN40-INVITES
 permit udp any eq 68 any eq 67
 deny ip 192.168.40.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip 192.168.40.0 0.0.0.255 any
!
ip access-list extended ACL-VLAN99-MGMT
 permit ip 192.168.99.0 0.0.0.255 any
!
! ---------- Router-on-a-stick ----------
interface GigabitEthernet0/0
 description TRUNK-vers-SW1
 no ip address
 no shutdown
!
interface GigabitEthernet0/0.10
 description VLAN10-DIRECTION
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip nat inside
 ip access-group ACL-VLAN10-DIRECTION in
!
interface GigabitEthernet0/0.20
 description VLAN20-EMPLOYES
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip nat inside
 ip access-group ACL-VLAN20-EMPLOYES in
!
interface GigabitEthernet0/0.30
 description VLAN30-SERVEURS
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip nat inside
 ip access-group ACL-VLAN30-SERVEURS in
!
interface GigabitEthernet0/0.40
 description VLAN40-INVITES
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
 ip nat inside
 ip access-group ACL-VLAN40-INVITES in
!
interface GigabitEthernet0/0.99
 description VLAN99-MGMT
 encapsulation dot1Q 99
 ip address 192.168.99.1 255.255.255.0
 ip access-group ACL-VLAN99-MGMT in
!
! ---------- WAN + NAT (VLAN 99 volontairement exclu d'Internet) ----------
interface GigabitEthernet0/1
 description WAN-vers-R-ISP
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no shutdown
!
ip access-list standard NAT-LAN
 permit 192.168.10.0 0.0.0.255
 permit 192.168.20.0 0.0.0.255
 permit 192.168.30.0 0.0.0.255
 permit 192.168.40.0 0.0.0.255
ip nat inside source list NAT-LAN interface GigabitEthernet0/1 overload
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
! ---------- DHCP ----------
ip dhcp excluded-address 192.168.10.1 192.168.10.9
ip dhcp excluded-address 192.168.20.1 192.168.20.9
ip dhcp excluded-address 192.168.40.1 192.168.40.9
!
ip dhcp pool VLAN10
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 192.168.30.10
ip dhcp pool VLAN20
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.30.10
ip dhcp pool VLAN40
 network 192.168.40.0 255.255.255.0
 default-router 192.168.40.1
 dns-server 8.8.8.8
!
! ---------- Administration securisee (SSH depuis VLAN 99 uniquement) ----------
ip domain-name lab.local
username admin secret Lab@Secret2026
! Si PT refuse la commande suivante : tapez "crypto key generate rsa" puis 2048
crypto key generate rsa modulus 2048
ip ssh version 2
ip access-list standard MGMT-ONLY
 permit 192.168.99.0 0.0.0.255
login block-for 120 attempts 3 within 60
line vty 0 4
 login local
 transport input ssh
 access-class MGMT-ONLY in
 exec-timeout 5 0
line console 0
 password Cons@2026
 login
 logging synchronous
 exec-timeout 5 0
end
write memory
