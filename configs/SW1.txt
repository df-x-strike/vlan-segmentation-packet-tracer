enable
configure terminal
hostname SW1
no ip domain-lookup
service password-encryption
enable secret Lab@Secret2026
banner motd # Acces reserve au personnel autorise #
vtp mode transparent
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,99 priority 4096
!
vlan 10
 name DIRECTION
vlan 20
 name EMPLOYES
vlan 30
 name SERVEURS
vlan 40
 name INVITES
vlan 99
 name MGMT
vlan 666
 name PARKING-PORTS-INUTILISES
vlan 999
 name NATIF-INUTILISE
!
! ---------- Trunks (VLAN autorises limites, natif 999, DTP desactive) ----------
interface GigabitEthernet0/1
 description TRUNK-vers-R1
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,99
 switchport nonegotiate
!
interface GigabitEthernet0/2
 description TRUNK-vers-SW2
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,40,99
 switchport nonegotiate
!
! ---------- Ports d'acces ----------
interface FastEthernet0/1
 description SRV-INT
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
!
interface FastEthernet0/2
 description PC-ADMIN
 switchport mode access
 switchport access vlan 99
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
!
! ---------- Ports inutilises : parking + shutdown ----------
interface range FastEthernet0/3 - 24
 description INUTILISE
 switchport mode access
 switchport access vlan 666
 shutdown
!
! ---------- Administration (SVI VLAN 99 + SSH) ----------
interface Vlan99
 description MGMT-SW1
 ip address 192.168.99.2 255.255.255.0
 no shutdown
ip default-gateway 192.168.99.1
!
ip domain-name lab.local
username admin secret Lab@Secret2026

line vty 0 15
